\begin{abstract}
We prove that a real additive function which decreases on a set of
natural density zero is a nonnegative constant multiple of the logarithm.
This settles a conjecture of Erd\H{o}s from 1946, listed as Problem~1122
on the Erd\H{o}s Problems website. The main step is to deduce finite
concentration without a growth assumption on the values at higher prime powers.
A truncated normalization and a bounded clipping reduce this step to
Mangerel's first-moment theorem for short intervals, Ruzsa's
second-moment estimate, and Elliott's fourth-moment inequality.
\end{abstract}

\section{Introduction}

Throughout, $\N=\{1,2,\ldots\}$.
A function $f\colon\N\to\R$ is \emph{additive} if
$f(ab)=f(a)+f(b)$ whenever $(a,b)=1$. Erd\H{o}s proved that an additive
function which is nondecreasing on $\N$ must be a constant multiple of
$\log n$. He conjectured that the same conclusion holds when the set of
decreases has natural density zero \cite[p.~3]{Erdos}. The conjecture is
also recorded as Erd\H{o}s Problem~1122 \cite{Problem1122}.

\begin{theorem}\label{thm:main}
Let $f\colon\N\to\R$ be additive. If
\[
 D_f(X):=\#\{n\le X:f(n+1)<f(n)\}=o(X),
\]
then there is a constant $c\ge0$ such that $f(n)=c\log n$ for every
$n\in\N$.
\end{theorem}

A Lean~4 formalization of the argument, treating the cited results of
Mangerel, Ruzsa, Erd\H{o}s, and Hildebrand as hypotheses, is available
online.\footnote{\url{https://github.com/FireflySentinel/erdos-1122}.}

Erd\H{o}s proved that $f(n+1)-f(n)=o(1)$ forces $f=c\log$.
K\'atai obtained the same conclusion from
$X^{-1}\sum_{n\le X}|f(n+1)-f(n)|\to0$ \cite{Katai}.
Wirsing gave another proof \cite{Wirsing};
see also Elliott \cite[Ch.~11, pp.~244--247]{ElliottBook}.
Hildebrand proved the density-one version \cite{Hildebrand}.
Mangerel obtained an approximate logarithmic representation under the
present density hypothesis \cite[Theorem~1.9]{Mangerel}, and the exact
conclusion for completely additive functions under additional conditions
on prime values and decreases \cite[Corollary~1.7]{Mangerel}.

We say that $f$ has \emph{finite concentration} if there are constants
$d>0$ and $R<\infty$ such that, for arbitrarily large $X$, some interval
of length $R$ contains at least $dX$ of the values $f(n)$, $n\le X$.
The interval may depend on $X$. Erd\H{o}s's Theorem~V
\cite{Erdos} makes finite concentration equivalent to the existence
of $c\in\R$ such that
\begin{equation}\label{eq:classical-concentration}
 \sum_p\frac{\min\{(f(p)-c\log p)^2,1\}}p<\infty.
\end{equation}
Under \eqref{eq:classical-concentration}, Hildebrand gives a symmetric
limiting law for $f(n+1)-f(n)$; the density hypothesis and his corollary
then force $f=c\log$. Thus it suffices to prove finite concentration.
Assuming the contrary, we normalize $f$ modulo a logarithmic term to fix
its truncated quadratic prime mass. Ruzsa gives a positive lower bound
for the variance after clipping, while Elliott supplies the fourth-moment
control needed to combine Mangerel's short-interval estimate with the
density hypothesis and obtain an incompatible upper bound.

\subsection*{Notation}
Letters $p,q$ denote primes. We write
\[
 \E_X F=\frac1X\sum_{n\le X}F(n),\qquad
 \mathcal A_H F(n)=\frac1H\sum_{j=0}^{H-1}F(n-j)\quad(n\ge H).
\]
For strongly additive $z(n)=\sum_{p\mid n}a_p$,
put $\mu_z(X)=\sum_{p\le X}a_p/p$.
All implicit constants are absolute unless a dependence is indicated.
The constants $K$ and $M$ used in the proof are fixed before $X$ tends
to infinity. In statements about a family $z_X$, the prime coefficients
may depend on $X$.

\section{Moment estimates and short intervals}

For an additive function $z$, define
\[
 A_0(z,X)=\sum_{p^k\le X}\frac{z(p^k)}{p^k}\left(1-\frac1p\right),
 \qquad
 B_2(z,X)^2=\sum_{p^k\le X}\frac{|z(p^k)|^2}{p^k}.
\]
Ruzsa's second-moment estimate \cite{Ruzsa}, in the form recorded by
Mangerel \cite[Lemma~3.3]{Mangerel}, gives, for all sufficiently large $X$
and uniformly in $z$,
\begin{equation}\label{eq:ruzsa}
 \E_X|z-A_0(z,X)|^2
 \asymp\inf_{b\in\R}\left(b^2+
       \sum_{p^k\le X}\frac{|z(p^k)-b\log(p^k)|^2}{p^k}\right).
\end{equation}
The upper bound with $b=0$ is the Tur\'an--Kubilius inequality, valid
for all $X\ge2$ \cite[Ch.~III.3]{Tenenbaum}.
For a strongly additive function $z(n)=\sum_{p\mid n}a_p$, summing
the geometric series gives
\[
 A_0(z,X)-\mu_z(X)=-\sum_{p\le X}a_p p^{-k_p},
 \qquad k_p=\min\{k\ge1:p^k>X\}.
\]
Thus $|A_0(z,X)-\mu_z(X)|\le X^{-1}\sum_{p\le X}|a_p|
\le(\sum_{p\le X}a_p^2/p)^{1/2}$ by Cauchy--Schwarz and
$\sum_{p\le X}p\le X^2$. When $|a_p|\le L$, the same difference
is at most $L\pi(X)/X=o(1)$.
Elliott's unweighted center differs from $\mu_z(X)$ by at most
$\sum_{p\le X}|a_p|/[p(p-1)]\ll(\sum_{p\le X}a_p^2/p)^{1/2}$,
so his fourth-moment inequality \cite[Theorem~1]{Elliott} gives
\begin{equation}\label{eq:fourth}
 \E_X|z-\mu_z(X)|^4
 \ll \left(\sum_{p\le X}\frac{a_p^2}{p}\right)^2
       +\sum_{p\le X}\frac{a_p^4}{p},\qquad X\ge2.
\end{equation}
Also
$B_2(z,X)^2\le2\sum_{p\le X}a_p^2/p$. Combining Ruzsa's upper
bound with \eqref{eq:fourth}, if $|a_p|\le L$ and
$\sum_{p\le X}a_p^2/p\le V$, then
\begin{equation}\label{eq:bounded-moments}
 \E_X|z-\mu_z(X)|^2\ll V,\qquad
 \E_X|z-\mu_z(X)|^4\ll V^2+L^2V.
\end{equation}

\begin{lemma}\label{lem:short}
Let $z_X$ be strongly additive functions satisfying
\[
 |z_X(p)|\le L,\qquad
 \sum_{p\le X}\frac{z_X(p)^2}{p}\le V
\]
for fixed $L,V<\infty$. Then
\begin{equation}\label{eq:short-limit}
 \lim_{H\to\infty}\limsup_{X\to\infty}
 \frac1X\sum_{H\le n\le X}
 |\mathcal A_Hz_X(n)-\mu_{z_X}(X)|^2=0,
\end{equation}
where $H$ runs through the positive integers.
\end{lemma}

\begin{proof}
For $Y\le X$, write
\[
 d_Y=\frac2Y\sum_{Y/2<n\le Y}z_X(n),\qquad
 F_Y(n)=\mathcal A_Hz_X(n)-d_Y.
\]
Mangerel's Theorem~1.1 \cite{Mangerel} gives, for every integer
$10\le H\le Y/100$,
\begin{equation}\label{eq:mangerel}
 \frac2Y\sum_{Y/2<n\le Y}|F_Y(n)|
 \ll\left(\sqrt{\frac{\log\log H}{\log H}}
                +(\log Y)^{-1/800}\right)\sqrt{2V}.
\end{equation}
Here $B_2(z_X,Y)^2\le 2V$, and the estimate is uniform over additive
functions, so it applies to the family $z_X$.

Counting multiples of each prime gives
\begin{equation}\label{eq:dyadic-center}
 d_Y-\mu_{z_X}(Y)
 =\sum_{p\le Y}z_X(p)
 \left(\frac2Y\left(\left\lfloor\frac Yp\right\rfloor
       -\left\lfloor\frac Y{2p}\right\rfloor\right)-\frac1p\right)
 =O\left(\frac{L\pi(Y)}Y\right).
\end{equation}
Jensen's inequality and \eqref{eq:bounded-moments} at $Y$ imply
\begin{equation}\label{eq:discrepancy-fourth}
 \frac2Y\sum_{Y/2<n\le Y}|F_Y(n)|^4
 \ll V^2+L^2V+O_L((\log Y)^{-4}).
\end{equation}
Indeed, with $\mu=\mu_{z_X}(Y)$, Jensen's inequality gives
\[
 |\mathcal A_H(z_X-\mu)(n)|^4
 \le \mathcal A_H(|z_X-\mu|^4)(n).
\]
Since $H\le Y/100<Y/2$, each index in these windows lies in $[1,Y]$
and occurs at most $H$ times when we sum over $Y/2<n\le Y$.
Equation \eqref{eq:dyadic-center} bounds the center shift.
H\"older's inequality now gives
\[
 \frac2Y\sum|F_Y|^2
 \le\left(\frac2Y\sum|F_Y|\right)^{2/3}
       \left(\frac2Y\sum|F_Y|^4\right)^{1/3}.
\]
Together with \eqref{eq:mangerel} and
\eqref{eq:discrepancy-fourth}, this proves the required iterated limit
on each fixed dyadic band $Y\asymp X$.

To extend this conclusion to the whole interval, fix $\eta=2^{-r}$.
On each of the $r$ dyadic bands covering $(\eta X,X]$, we have
\[
 |\mu_{z_X}(X)-\mu_{z_X}(Y)|
 \le L\sum_{Y<p\le X}\frac1p=o(1).
\]
The omitted initial segment $H\le n\le\eta X$ contributes
$O_{L,V}(\sqrt\eta)$ to \eqref{eq:short-limit}: apply
Cauchy--Schwarz and Jensen, using the fourth-moment bound
\eqref{eq:bounded-moments} at $X$. Take $X\to\infty$, then
$H\to\infty$; since $\eta$ is arbitrary, \eqref{eq:short-limit} follows.
\end{proof}

\section{A weighted estimate for a set of primes}

\begin{lemma}\label{lem:tail-weight}
Suppose $0<M<1/4$ and $\mathcal T$ is a set of primes at most $X$
such that $\sum_{p\in\mathcal T}1/p\le M$. Then, uniformly in
$\mathcal T$,
\begin{equation}\label{eq:tail-weight}
 \sum_{p\in\mathcal T}\frac1p
       \sum_{X/p<q\le X}\frac1q
 \ll M\log(2/M)+o_{X\to\infty}(1).
\end{equation}
Here $M$ is fixed in the term $o(1)$.
\end{lemma}

\begin{proof}
We use Mertens's prime harmonic-sum formula and the standard
prime-counting bound; see \cite{Tenenbaum}. For $p\le X^{1-M}$, the inner sum is at most
$\log(1/M)+o(1)$. These primes
contribute at most $M\log(1/M)+o(1)$.

For $p>X^{1-M}$, discard the restriction $p\in\mathcal T$ and
interchange the sums. The terms with $q>X^M$ are bounded by
\[
 \left(\sum_{X^{1-M}<p\le X}\frac1p\right)
 \left(\sum_{X^M<q\le X}\frac1q\right)
 \ll M\log(2/M)+o(1).
\]
For $q\le X^M$, the condition $X/p<q$ implies $p>X/q$.
The bound $\pi(t)\ll t/\log t$ and partial summation give,
uniformly for $2\le q\le X^M$,
\[
 \sum_{X/q<p\le X}\frac1p\ll\frac{\log q}{\log X}.
\]
Their total is $\ll(\log X)^{-1}\sum_{q\le X^M}(\log q)/q\ll M$,
proving \eqref{eq:tail-weight}.
\end{proof}

\section{Normalization}

In the remainder of the proof, suppose that $f$ does not have finite
concentration. For $s>0$ define
\begin{equation}\label{eq:W}
 W_X(s)=\min_{c\in\R}
 \left\{c^2+\sum_{p\le X}
       \frac{\min\{(f(p)/s-c\log p)^2,1\}}p\right\}.
\end{equation}

\begin{lemma}\label{lem:normalization}
For each fixed $M>0$ and all sufficiently large $X$, there are
$s_X>0$ and a minimizer $c_X$ in \eqref{eq:W} such that
$W_X(s_X)=M$. Then necessarily $s_X\to\infty$, and any corresponding
minimizers satisfy $c_X\to0$. Put $u_p=f(p)/s_X-c_X\log p$ for
$p\le X$. Then
\begin{align}
 \sum_{p\le X}\frac{\min\{u_p^2,1\}}p
 &=M-c_X^2=M-o(1),\label{eq:mass}\\
 \sum_{\substack{p\le X\\|u_p|<1}}
       \frac{u_p\log p}{p}&=c_X.\label{eq:stationarity}
\end{align}
No $u_p$ has absolute value exactly $1$.
\end{lemma}

\begin{proof}
The expression in braces in \eqref{eq:W} is continuous and coercive
as a function of $c$, so its minimum is attained. Every minimizer
satisfies $c^2\le\sum_{p\le X}1/p$, by comparison with $c=0$.
The minimum can therefore be taken over a fixed compact interval,
independent of $s$, and is continuous in $s>0$.
On writing $\lambda=sc$, the expression
becomes
\[
 \frac{\lambda^2}{s^2}
 +\sum_{p\le X}\frac{\min\{(f(p)-\lambda\log p)^2/s^2,1\}}p.
\]
It follows that $W_X(s)$ is nonincreasing in $s$, and it tends to
zero as $s\to\infty$ for fixed $X$.

For each fixed $s>0$, $W_X(s)\to\infty$. Otherwise
some unbounded sequence of $X$ would have bounded $W_X(s)$ and
bounded minimizers. Pass to a subsequence along which the minimizers
converge to $c$. Fatou's lemma gives
\[
 \sum_p\frac{\min\{(f(p)/s-c\log p)^2,1\}}p<\infty.
\]
Rescaling by the fixed number $s$ preserves convergence of the
truncated square sum. Erd\H{o}s's Theorem~V would therefore give
finite concentration of $f$, contrary to the assumption.

Continuity now supplies $s_X$ with $W_X(s_X)=M$. Monotonicity and
the preceding claim imply $s_X\to\infty$ for every such choice. We have
$|c_X|\le\sqrt M$. If a subsequential limit of $c_X$ were a
nonzero number $c$, Fatou's lemma on the fixed primes would give
\[
 \sum_p\frac{\min\{(c\log p)^2,1\}}p\le M,
\]
which is impossible. Thus $c_X\to0$, and \eqref{eq:mass} follows.

It remains to prove the stationary identity and exclude equality at
the truncation threshold. Let $G(c)$ denote the expression in braces
in \eqref{eq:W}, with $s=s_X$. For any subset $J$ of the primes at most
$X$, the quadratic
\[
 Q_J(c)=c^2+\sum_{p\in J}\frac{(f(p)/s_X-c\log p)^2}{p}
             +\sum_{\substack{p\le X\\p\notin J}}\frac1p
\]
lies above $G(c)$. Choose $J=\{p\le X:|u_p|<1\}$. Then
$Q_J(c_X)=G(c_X)$, so $c_X$ also minimizes $Q_J$. The equation
$Q_J'(c_X)=0$ is exactly \eqref{eq:stationarity}.
If $|u_p|=1$ for some $p$, the quadratic $Q_{J\cup\{p\}}$ also
touches $G$ at $c_X$ and must have derivative zero there, contradicting
\[
 Q_{J\cup\{p\}}'(c_X)-Q_J'(c_X)
 =-\frac{2u_p\log p}{p}\ne0.\qedhere
\]
\end{proof}

\section{Clipping the normalized function}

Fix $K>1$ and $0<M<1/4$, and use Lemma~\ref{lem:normalization}.
Let $\clip(t)=\max\{-K,\min\{t,K\}\}$ and set
\begin{align*}
 \mathcal T&=\{p\le X:|u_p|>1\},&
 S_0(n)&=\sum_{\substack{p\mid n\\p\notin\mathcal T}}u_p,\\
 a_X&=\sum_{\substack{p\le X\\p\notin\mathcal T}}\frac{u_p}{p},&
 S(n)&=S_0(n)-a_X.
\end{align*}
Coefficients at primes greater than $X$ are set to zero. Define
\begin{equation}\label{eq:clipped-and-Z}
 \begin{split}
 z_X(n)&=\sum_{p\mid n}\clip(u_p),\qquad Z(n)=z_X(n)-a_X,\\
 g_X(n)&=\clip\left(\frac{f(n)}{s_X}-c_X\log n-a_X\right).
 \end{split}
\end{equation}
Thus $g_X$ clips the value of the additive function, whereas $z_X$
clips the prime contributions individually.

\begin{lemma}\label{lem:clipping}
There is an absolute constant $C$ such that
\begin{equation}\label{eq:clipping-bound}
 \limsup_{X\to\infty}\E_X|g_X-Z|^2
 \le C\left(\frac M{K^2}+M^2\log(2/M)+K^2M^2\right).
\end{equation}
\end{lemma}

\begin{proof}
The small coefficients have quadratic mass at most $M$, and
$\sum_{p\in\mathcal T}1/p\le M$. Hence
\begin{equation}\label{eq:S-fourth}
 \E_X S^4\ll M+M^2.
\end{equation}
Put $\omega_{\mathcal T}(n)=\#\{p\in\mathcal T:p\mid n\}$. Counting multiples of
distinct primes gives
\begin{equation}\label{eq:tail-count}
 \E_X\omega_{\mathcal T}(\omega_{\mathcal T}-1)
 =\sum_{\substack{p,q\in\mathcal T\\p\ne q}}
       \frac1X\left\lfloor\frac X{pq}\right\rfloor
 \le M^2.
\end{equation}
Moreover,
\begin{equation}\label{eq:mixed-moment}
 \E_X\bigl(S^2\ind_{\omega_{\mathcal T}\ge1}\bigr)
 \ll M^2\log(2/M)+o(1).
\end{equation}
For $p\in\mathcal T$, the coefficient of $p$ in $S_0$ is zero,
so $S_0(pm)=S_0(m)$ for every $m$, including multiples of $p$.
Writing $y=X/p$, for $y\ge2$ equation \eqref{eq:bounded-moments} gives
\begin{align*}
 \E_X\bigl(S^2\ind_{p\mid n}\bigr)
 &=\frac1p\frac1y\sum_{m\le y}|S_0(m)-a_X|^2\\
 &\ll\frac1p\left(M+
           |a_X-\mu_{S_0}(y)|^2\right).
\end{align*}
For $1\le y<2$, the same bound follows directly from
$S_0(1)=\mu_{S_0}(y)=0$. By Cauchy--Schwarz,
\[
 |a_X-\mu_{S_0}(X/p)|^2
 \le M\sum_{X/p<q\le X}\frac1q.
\]
Hence, using $\ind_{\omega_{\mathcal T}\ge1}\le\sum_{p\in\mathcal T}\ind_{p\mid n}$,
\begin{align*}
 \E_X\bigl(S^2\ind_{\omega_{\mathcal T}\ge1}\bigr)
 &\ll M\sum_{p\in\mathcal T}\frac1p
 +M\sum_{p\in\mathcal T}\frac1p
       \sum_{X/p<q\le X}\frac1q\\
 &\ll M^2+M^2\log(2/M)+o(1),
\end{align*}
by Lemma~\ref{lem:tail-weight}. This proves \eqref{eq:mixed-moment}.

Set
\[
 g_X^*(n)=\clip\left(S(n)+
                  \sum_{\substack{p\mid n\\p\in\mathcal T}}u_p\right).
\]
If $\omega_{\mathcal T}(n)=0$, the difference $g_X^*(n)-Z(n)$ vanishes for $|S(n)|\le K$
and otherwise has absolute value at most $|S(n)|$. Its squared
contribution is at most $\E_X S^4/K^2$.
If $\omega_{\mathcal T}(n)=1$, with unique tail value $v$, the Lipschitz property of
$\clip$ gives
\[
 |\clip(S+v)-S-\clip(v)|\le2|S|.
\]
If $\omega_{\mathcal T}(n)\ge2$, then
\[
 |g_X^*(n)-Z(n)|\le |S(n)|+K(\omega_{\mathcal T}(n)+1)
 \le |S(n)|+\frac32K\omega_{\mathcal T}(n),
\]
and hence
\[
 |g_X^*(n)-Z(n)|^2\le C S(n)^2+
 C K^2\omega_{\mathcal T}(n)(\omega_{\mathcal T}(n)-1).
\]
Equations \eqref{eq:S-fourth}--\eqref{eq:mixed-moment} therefore imply
\begin{equation}\label{eq:strong-clipping}
 \E_X|g_X^*-Z|^2
 \ll\frac M{K^2}+M^2\log(2/M)+K^2M^2+o(1).
\end{equation}

To compare $g_X$ and $g_X^*$, suppose $p^k\Vert n$ with $k\ge2$. Its
contribution to the difference between the two unclipped expressions is
\[
 \frac{f(p^k)-f(p)}{s_X}-c_X(k-1)\log p.
\]
This tends to zero for each fixed prime power. Moreover,
$\sum_{p,k\ge2}p^{-k}<\infty$. Given $\varepsilon>0$, choose a finite set $\mathcal F$ of prime
powers such that $\sum_{p^k\notin\mathcal F,\,k\ge2}p^{-k}<\varepsilon$.
Then, uniformly in $X$,
\[
 \frac1X\#\{n\le X:\text{ some }p^k\notin\mathcal F,\ k\ge2,
                    \text{ divides }n\}<\varepsilon.
\]
Outside this exceptional set, only the fixed prime powers in
$\mathcal F$ contribute, and their total contribution tends uniformly
to zero. The Lipschitz bound and $|g_X|,|g_X^*|\le K$ therefore give
\[
 \limsup_{X\to\infty}\E_X|g_X-g_X^*|^2\le4K^2\varepsilon.
\]
Letting $\varepsilon\downarrow0$ yields $\E_X|g_X-g_X^*|^2=o(1)$;
combine this with \eqref{eq:strong-clipping}.
\end{proof}

\begin{lemma}\label{lem:variance-lower}
There are absolute constants $c_0,C_0>0$ such that
\begin{equation}\label{eq:variance-lower}
 \liminf_{X\to\infty}\E_X|z_X-\mu_{z_X}(X)|^2
 \ge c_0M-C_0K^2M^2.
\end{equation}
\end{lemma}

\begin{proof}
Write $v_p=\clip(u_p)$. Since $K>1$, \eqref{eq:mass} implies
\begin{equation}\label{eq:clipped-mass}
 M-o(1)\le\sum_{p\le X}\frac{v_p^2}{p}\le K^2M.
\end{equation}
On $p\notin\mathcal T$ we have $v_p=u_p$, so
\eqref{eq:stationarity} gives
\begin{equation}\label{eq:projection-numerator}
 \left|\sum_{p\le X}\frac{v_p\log p}{p}\right|
 \le |c_X|+KM\log X.
\end{equation}
By partial summation from Mertens's first theorem,
$\sum_{p\le X}(\log p)^2/p\sim\tfrac12(\log X)^2$.
The quadratic projection identity gives
\begin{align}
 \inf_{b\in\R}\sum_{p\le X}\frac{(v_p-b\log p)^2}{p}
 &=\sum_{p\le X}\frac{v_p^2}{p}
   -\frac{\bigl(\sum_{p\le X}v_p\log p/p\bigr)^2}
          {\sum_{p\le X}(\log p)^2/p}\notag\\
 &\ge M-C K^2M^2-o(1).\label{eq:projection}
\end{align}
Apply \eqref{eq:ruzsa}, dropping the nonnegative $b^2$ term and the
terms with $k\ge2$ from its right-hand side.
Since $|v_p|\le K$, the center estimate in Section~2 gives
$|A_0(z_X,X)-\mu_{z_X}(X)|\le K\pi(X)/X=o(1)$.
The second moments are uniformly bounded by
\eqref{eq:bounded-moments} and \eqref{eq:clipped-mass}; changing the
center consequently changes the second moment by $o(1)$.
Combining this with \eqref{eq:projection} proves
\eqref{eq:variance-lower}.
\end{proof}

\section{Proof of Theorem~\ref{thm:main}}

Assume that $f$ has no finite concentration and take $g_X$ from
\eqref{eq:clipped-and-Z}. If $f(n+1)\ge f(n)$, monotonicity and Lipschitz
continuity of $\clip$ give
$(g_X(n)-g_X(n+1))_+\le |c_X|\log(1+1/n)$, where $t_+=\max\{t,0\}$.
Using $|g_X|\le K$ and telescoping the signed increments gives
\begin{equation}\label{eq:variation}
 \sum_{n<X}|g_X(n+1)-g_X(n)|
 \le2K+4KD_f(X)+2|c_X|\log X=o(X).
\end{equation}
This is where clipping enters: a decrease of $f$ may be arbitrarily
large, but changes $g_X$ by at most $2K$.
For each fixed $H$, every difference $g_X(n)-g_X(n-j)$ is bounded by
the sum of the $j$ intervening absolute increments. Since $|g_X|\le K$,
\eqref{eq:variation} yields
\begin{equation}\label{eq:clipped-short}
 \frac1X\sum_{H\le n\le X}|g_X(n)-\mathcal A_H g_X(n)|^2\longrightarrow0.
\end{equation}

Let $m_X=\mu_{z_X}(X)-a_X$. Then $Z-m_X=z_X-\mu_{z_X}(X)$, and
\begin{equation}\label{eq:four-terms}
 Z-m_X=(Z-g_X)+(g_X-\mathcal A_H g_X)
            +\mathcal A_H(g_X-Z)+(\mathcal A_H Z-m_X).
\end{equation}
The square of the right-hand side is at most four times the sum of
the four squared terms. Jensen's inequality gives the contraction
\[
 \frac1X\sum_{H\le n\le X}|\mathcal A_H(g_X-Z)(n)|^2
 \le\frac1X\sum_{n\le X}|g_X(n)-Z(n)|^2.
\]
Lemma~\ref{lem:short} applies to $z_X$, whose prime coefficients are
bounded by $K$ with quadratic mass at most $K^2M$. The first $H$
indices contribute $o(1)$ by \eqref{eq:bounded-moments} and
Cauchy--Schwarz. Taking $X\to\infty$ and then $H\to\infty$ in
\eqref{eq:four-terms}, and using \eqref{eq:clipped-short}, gives
\begin{align}
 \limsup_X\E_X|z_X-\mu_{z_X}(X)|^2
 &\le8\limsup_X\E_X|g_X-Z|^2\notag\\
 &\le C_1\left(\frac M{K^2}+M^2\log(2/M)+K^2M^2\right)
       \label{eq:variance-upper}
\end{align}
for an absolute constant $C_1$.

Choose $K>1$ so large that $C_1/K^2<c_0/4$, where $c_0$ is from
Lemma~\ref{lem:variance-lower}. Next choose $0<M<1/4$ so small that
\[
 C_1M\log(2/M)+(C_1+C_0)K^2M<c_0/4.
\]
With these fixed choices, \eqref{eq:variance-upper} contradicts
\eqref{eq:variance-lower}. Hence $f$ has finite concentration.

Erd\H{o}s's Theorem~V now gives
\eqref{eq:classical-concentration}. Hildebrand's Theorem~1 gives a limiting distribution for $f(n+1)-f(n)$
whose characteristic function \cite[(1.4)]{Hildebrand} is a product of
real factors, hence the law is symmetric. For every $\varepsilon>0$,
$\#\{n\le X:f(n+1)-f(n)<-\varepsilon\}\le D_f(X)=o(X)$.
The limiting law assigns no mass to $(-\infty,0)$; by symmetry it is
therefore $\delta_0$.
Hence $f(n+1)-f(n)\to0$ in density, and a standard diagonal argument
yields a density-one set along which the differences tend to zero.
Hildebrand's corollary \cite[pp.~258--259]{Hildebrand}
now gives $f(n)=c\log n$ for some real $c$ and all $n$.
Finally, $c<0$ would give a decrease at every integer, so $c\ge0$.\qed
